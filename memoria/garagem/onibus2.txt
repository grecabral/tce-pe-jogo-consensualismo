<script>
const canvas = document.getElementById('gameCanvas');
const ctx = canvas.getContext('2d', { alpha: false });

let audioCtx, motorOsc, motorGain;
function iniciarAudio() {
    if (audioCtx) return;
    audioCtx = new (window.AudioContext || window.webkitAudioContext)();
    motorOsc = audioCtx.createOscillator();
    motorGain = audioCtx.createGain();
    motorOsc.type = 'sawtooth';
    motorOsc.connect(motorGain);
    motorGain.connect(audioCtx.destination);
    motorGain.gain.setValueAtTime(0, audioCtx.currentTime);
    motorOsc.start();
}

function playSound(freq, type, duration, vol) {
    if (!audioCtx) return;
    const osc = audioCtx.createOscillator();
    const g = audioCtx.createGain();
    osc.type = type; osc.connect(g); g.connect(audioCtx.destination);
    osc.frequency.setValueAtTime(freq, audioCtx.currentTime);
    g.gain.setValueAtTime(vol, audioCtx.currentTime);
    g.gain.exponentialRampToValueAtTime(0.01, audioCtx.currentTime + duration);
    osc.start(); osc.stop(audioCtx.currentTime + duration);
}

canvas.width = 600; canvas.height = window.innerHeight;

const imgPista = new Image(); imgPista.src = '../assets/pista.jpg';
const imgBus = new Image(); imgBus.src = '../assets/onibus.png';

let gameActive = true, timeLeft = 60, passageirosTotais = 0;
// 2. Velocidade diminuída em 10% (de 11.2 para 10.0)
let velocidadeCruzeiro = 10.0, velocidadeAtual = 10.0, pistaY = 0;
let mouseX = 300, estado = "DIRIGINDO", lastParadaTime = Date.now();

// 4. Dimensões ajustadas para manter o ônibus na pista
const bus = { x: 270, y: canvas.height - 180, w: 65, h: 130 };
const buracos = [];
const parada = { x: 440, y: -500, w: 80, h: 320, ativa: false, palavra: "" };

// 1. Novos nomes para os obstáculos
const palavrasBuraco = ["Atrasos", "Demora", "Má fé", "Impasse", "Imposição", "Hostilidade", "Intimidação", "Coação"];
const palavrasParada = ["HORIZONTALIDADE", "ESCUTA ATIVA", "MEDIAÇÃO", "DIÁLOGO", "CONSENSO"];
let paradaIndex = 0;

window.addEventListener('mousemove', (e) => mouseX = e.clientX - canvas.getBoundingClientRect().left);

const timerInterval = setInterval(() => {
    if (gameActive && estado !== "EMBARCANDO") {
        timeLeft--;
        document.getElementById('timer').innerText = Math.max(0, timeLeft);
        if (timeLeft <= 0) finalizar();
    }
}, 1000);

function finalizar() {
    gameActive = false;
    clearInterval(timerInterval);
    document.getElementById('tela-final').style.display = 'flex';
    
    // 3 e 5. Lógica de vitória por passageiros e narrativa
    const venceu = passageirosTotais >= 40;
    
    if(venceu) {
        document.getElementById('res-titulo').innerText = "MISSÃO CUMPRIDA!";
        document.getElementById('res-desc').innerHTML = `
            O transporte chegou a tempo! Graças à sua gestão baseada no <strong>Consensualismo</strong>, 
            os conflitos foram superados e a cooperação permitiu que ${passageirosTotais} cidadãos 
            chegassem ao atendimento médico. Você provou que o diálogo destrava cidades!
        `;
        playSound(800, 'sine', 1, 0.2);
    } else {
        document.getElementById('res-titulo').innerText = "FALTOU DIÁLOGO";
        document.getElementById('res-titulo').style.color = "#ef4444";
        document.getElementById('res-desc').innerHTML = `
            Muitos impasses e poucos acordos. Apenas ${passageirosTotais} passageiros conseguiram embarcar. 
            Sem o poder do consenso, a gestão pública trava e a população sofre com a demora.
        `;
        playSound(200, 'sine', 1, 0.2);
    }
}

function update() {
    if (!gameActive) return;

    // 3. Ganha o jogo imediatamente ao atingir a meta
    if (passageirosTotais >= 40) {
        finalizar();
        return;
    }

    if (motorGain) {
        motorGain.gain.setTargetAtTime(0.04, audioCtx.currentTime, 0.1);
        motorOsc.frequency.setTargetAtTime(40 + (velocidadeAtual * 6), audioCtx.currentTime, 0.1);
    }

    // 4. Limita o movimento do ônibus para não sair da imagem da pista
    let limiteEsquerdo = 130;
    let limiteDireito = canvas.width - 130 - bus.w;
    bus.x += (mouseX - bus.x - bus.w/2) * 0.25;
    if (bus.x < limiteEsquerdo) bus.x = limiteEsquerdo;
    if (bus.x > limiteDireito) bus.x = limiteDireito;

    if (estado === "DIRIGINDO" && Date.now() - lastParadaTime > 7000) {
        estado = "PARANDO"; parada.y = -400; parada.ativa = true;
        parada.palavra = palavrasParada[paradaIndex];
        paradaIndex = (paradaIndex + 1) % palavrasParada.length;
    }

    if (estado === "PARANDO") {
        parada.y += velocidadeAtual;
        const naZona = bus.x > 380 && Math.abs((bus.y + 65) - (parada.y + 160)) < 100;
        if (naZona) {
            velocidadeAtual = Math.max(0, velocidadeAtual - 0.9);
            if (velocidadeAtual === 0) {
                estado = "EMBARCANDO";
                document.getElementById('palavra-parada').innerText = `Parada: ${parada.palavra}`;
                document.getElementById('container-embarque').style.display = 'block';
                embarcar(0);
            }
        } else { velocidadeAtual = Math.max(4, velocidadeAtual - 0.05); }

        if (parada.y > canvas.height && estado === "PARANDO") {
            estado = "DIRIGINDO"; velocidadeAtual = velocidadeCruzeiro;
            parada.ativa = false; lastParadaTime = Date.now();
        }
    }

    pistaY += velocidadeAtual;
    if (pistaY >= canvas.height) pistaY = 0;

    buracos.forEach((b, i) => {
        b.y += velocidadeAtual;
        if (b.y + b.h > bus.y && b.y < bus.y + bus.h && b.x + b.w > bus.x && b.x < bus.x + bus.w) {
            playSound(80, 'square', 0.2, 0.1);
            buracos.splice(i, 1); 
            velocidadeAtual = 4;
        }
        if (b.y > canvas.height) buracos.splice(i, 1);
    });

    if (estado === "DIRIGINDO") {
        if (velocidadeAtual < velocidadeCruzeiro) velocidadeAtual += 0.12;
        if (Math.random() < 0.02) {
            // 4. Obstáculos apenas na área central da pista
            buracos.push({ 
                x: 180 + Math.random() * 200, y: -100, w: 60, h: 40, 
                txt: palavrasBuraco[Math.floor(Math.random() * palavrasBuraco.length)] 
            });
        }
    }
    document.getElementById('vel-val').innerText = Math.floor(velocidadeAtual * 7);
}

function embarcar(count) {
    if (count < 10) {
        count++; passageirosTotais++;
        playSound(600 + (count * 60), 'sine', 0.08, 0.1);
        document.getElementById('total-hud').innerText = passageirosTotais;
        document.getElementById('p-count').innerText = `${count} Passageiros`;
        setTimeout(() => embarcar(count), 200);
    } else {
        setTimeout(() => {
            estado = "DIRIGINDO"; velocidadeAtual = velocidadeCruzeiro;
            parada.ativa = false; lastParadaTime = Date.now();
            document.getElementById('container-embarque').style.display = 'none';
        }, 400);
    }
}

function draw() {
    ctx.clearRect(0, 0, canvas.width, canvas.height);
    ctx.drawImage(imgPista, 0, Math.floor(pistaY - canvas.height), canvas.width, canvas.height);
    ctx.drawImage(imgPista, 0, Math.floor(pistaY), canvas.width, canvas.height);

    if (parada.ativa) {
        ctx.fillStyle = "rgba(45, 212, 191, 0.3)";
        ctx.fillRect(parada.x, parada.y, parada.w, parada.h);
        ctx.save();
        ctx.translate(parada.x + parada.w/2, parada.y + parada.h/2);
        ctx.rotate(-Math.PI / 2);
        ctx.font = "bold 24px Arial";
        ctx.textAlign = "center";
        ctx.fillStyle = "#fff";
        ctx.fillText(parada.palavra, 0, 10);
        ctx.restore();
    }

    buracos.forEach(b => { 
        ctx.fillStyle = "rgba(239, 68, 68, 0.8)";
        ctx.beginPath(); ctx.ellipse(b.x+30, b.y+20, 35, 22, 0, 0, Math.PI*2); ctx.fill(); 
        ctx.font = "bold 18px Arial";
        ctx.fillStyle = "white"; ctx.textAlign = "center";
        ctx.fillText(b.txt, b.x + 30, b.y - 10);
    });

    ctx.drawImage(imgBus, Math.floor(bus.x), Math.floor(bus.y), bus.w, bus.h);
    update();
    requestAnimationFrame(draw);
}
imgPista.onload = () => draw();
</script>
